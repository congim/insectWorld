# 输入协议

> 定义客户端→服务端的操作请求、服务端→客户端的响应和推送消息格式，以及服务端校验规则。

---

## 操作类型枚举

| 操作类型 | 说明 | 请求频率限制 | 可预测 |
|---------|------|-------------|--------|
| move_entity | 移动实体到目标位置（含编队移动） | 1次/秒 | 是 |
| attack_entity | 攻击目标实体 | 1次/秒 | 否 |
| collect_resource | 采集资源 | 1次/秒 | 是 |
| build_entity | 建造建筑 | 无限制 | 否 |
| upgrade_entity | 升级建筑/科技 | 无限制 | 否 |
| recruit_unit | 招募兵种 | 无限制 | 否 |
| use_skill | 使用技能 | 按技能冷却 | 否 |
| teleport | 传送 | 按传送冷却 | 否 |
| create_alliance | 创建联盟 | 1次/分钟 | 否 |
| join_alliance | 加入联盟 | 1次/分钟 | 否 |
| leave_alliance | 退出联盟 | 无限制 | 否 |
| diplomacy_action | 外交操作 | 1次/10秒 | 否 |
| trade_action | 交易操作 | 1次/秒 | 否 |
| chat_send | 发送聊天 | 5次/秒 | 是 |
| convert_resource | 资源转换 | 1次/秒 | 否 |
| claim_welfare | 领取联盟福利 | 1次/分钟 | 否 |
| upgrade_alliance | 联盟升级 | 无限制 | 否 |
| claim_season_reward | 领取赛季奖励 | 1次/赛季 | 否 |
| query_game_event | 查询游戏事件 | 1次/秒 | 是 |
| manage_region | 区域管理（创建/销毁，运营） | 无限制 | 否 |
| trigger_terrain_change | 地形变更触发（运营） | 无限制 | 否 |
| config_reload | 触发配置热更（运营） | 1次/分钟 | 否 |

## 消息格式（Protobuf）

### 客户端请求

```protobuf
message ClientRequest {
  string request_id = 1;       // 请求唯一ID（UUID）
  int64 timestamp = 2;         // 客户端时间戳（毫秒）
  string player_id = 3;        // 玩家ID
  OperationType op_type = 4;   // 操作类型
  bytes op_data = 5;           // 操作数据（按操作类型序列化）
  int32 client_seq = 6;        // 客户端序列号（用于顺序保证）
  string predict_hash = 7;     // 预测状态哈希（可预测操作填写）
}

enum OperationType {
  MOVE_ENTITY = 0;
  ATTACK_ENTITY = 1;
  COLLECT_RESOURCE = 2;
  BUILD_ENTITY = 3;
  UPGRADE_ENTITY = 4;
  RECRUIT_UNIT = 5;
  USE_SKILL = 6;
  TELEPORT = 7;
  CREATE_ALLIANCE = 8;
  JOIN_ALLIANCE = 9;
  LEAVE_ALLIANCE = 10;
  DIPLOMACY_ACTION = 11;
  TRADE_ACTION = 12;
  CHAT_SEND = 13;
  CONFIG_RELOAD = 14;
  CONVERT_RESOURCE = 15;
  CLAIM_WELFARE = 16;
  UPGRADE_ALLIANCE = 17;
  CLAIM_SEASON_REWARD = 18;
  QUERY_GAME_EVENT = 19;
  MANAGE_REGION = 20;
  TRIGGER_TERRAIN_CHANGE = 21;
}
```

### 操作数据定义

```protobuf
// 移动请求
message MoveEntityData {
  string entity_id = 1;
  Position target_pos = 2;     // 目标坐标
  string movement_type = 3;    // 移动类型ID
  optional string formation_id = 4;  // 编队ID（可选）
}

// 攻击请求
message AttackEntityData {
  string attacker_id = 1;
  string target_id = 2;
  optional string skill_id = 3;  // 使用技能ID（可选）
}

// 采集请求
message CollectResourceData {
  string collector_id = 1;
  string target_id = 2;        // 资源点ID
}

// 建造请求
message BuildEntityData {
  string building_type = 1;
  Position position = 2;
  optional string owner_entity_id = 3;  // 所属实体（可选）
}

// 升级请求
message UpgradeEntityData {
  string entity_id = 1;
  string upgrade_target = 2;   // 升级目标（建筑等级/科技ID）
}

// 招募请求
message RecruitUnitData {
  string building_id = 1;      // 招募建筑
  string unit_type = 2;        // 兵种类型
  int32 count = 3;             // 招募数量
}

// 联盟操作
message AllianceActionData {
  string alliance_id = 1;
  optional string target_player_id = 2;  // 踢人/邀请时使用
  optional string diplomacy_state = 3;   // 外交操作时使用
  AllianceOpType action = 4;
}

enum AllianceOpType {
  CREATE = 0;
  JOIN = 1;
  LEAVE = 2;
  KICK = 3;
  INVITE = 4;
  DIPLOMACY = 5;
  UPGRADE = 6;
}

// 交易请求
message TradeActionData {
  string trade_type = 1;
  string target_player_id = 2;
  map<string, double> offer = 3;   // 提供的资源
  map<string, double> request = 4; // 请求的资源
}

// 聊天请求
message ChatSendData {
  string channel = 1;          // 频道：world/alliance/private
  string content = 2;
  optional string target_player_id = 3;  // 私聊时使用
}

// 资源转换请求
message ConvertResourceData {
  string conversion_rule_id = 1;  // 转换规则ID（对应config.conversion_rules）
  map<string, double> input_resources = 2;  // 输入资源
}

// 领取联盟福利请求
message ClaimWelfareData {
  string welfare_id = 1;       // 福利ID（对应config.alliance_welfare）
}

// 联盟升级请求
message UpgradeAllianceData {
  string alliance_id = 1;
}

// 领取赛季奖励请求
message ClaimSeasonRewardData {
  string season_id = 1;
  int32 rank_tier = 2;         // 奖励档位
}

// 查询游戏事件请求
message QueryGameEventData {
  optional string event_id = 1;  // 指定事件ID（可选，不填则查全部活跃事件）
}

// 区域管理请求（运营）
message ManageRegionData {
  string region_id = 1;
  RegionOp op = 2;             // 创建/销毁
  repeated Position cells = 3; // 格子集合（创建时填写）
  map<string, string> attributes = 4;  // 区域属性
}

enum RegionOp {
  CREATE_REGION = 0;
  DESTROY_REGION = 1;
}

// 地形变更触发请求（运营）
message TriggerTerrainChangeData {
  Position pos = 1;            // 目标坐标
  string new_terrain_id = 2;   // 新地形ID
}

message Position {
  int32 q = 1;                 // 六角格q坐标
  int32 r = 2;                 // 六角格r坐标
}
```

### 服务端响应

```protobuf
message ServerResponse {
  string request_id = 1;       // 对应请求的ID
  ResponseStatus status = 2;   // 处理状态
  bytes result_data = 3;       // 结果数据（按操作类型序列化）
  string error_code = 4;       // 错误码（失败时）
  string error_message = 5;    // 错误描述（失败时）
  int64 server_timestamp = 6;  // 服务端处理时间戳
  bool predict_confirmed = 7;  // 预测是否被确认（true=一致，false=需修正）
}

enum ResponseStatus {
  SUCCESS = 0;
  FAILED = 1;
  PENDING = 2;                 // 请求已接收，处理中（异步操作如建造）
}
```

### 服务端推送

```protobuf
message ServerPush {
  int64 timestamp = 1;         // 服务端时间戳
  int32 push_seq = 2;          // 推送序列号（用于丢包检测）
  PushType push_type = 3;      // 推送类型
  bytes push_data = 4;         // 推送数据
}

enum PushType {
  STATE_DELTA = 0;             // 状态增量

  SEASON_PHASE_CHANGE = 4;     // 赛季阶段变更
  CONFIG_RELOAD = 5;           // 配置热更通知
  ALLIANCE_NOTIFY = 6;         // 联盟通知
  CHAT_RECEIVE = 7;            // 聊天接收
  ERROR_NOTIFY = 8;            // 错误通知
}
```

## 服务端校验规则

### 校验流程

```
收到客户端请求
  │
  ├─ 1. 频率校验：检查该玩家该操作的请求频率
  │    └─ 超限 → 返回 FAILED, error_code="RATE_LIMIT"
  │
  ├─ 2. 身份校验：验证 player_id 与连接身份匹配
  │    └─ 不匹配 → 返回 FAILED, error_code="AUTH_MISMATCH"
  │
  ├─ 3. 权限校验：检查玩家是否有权执行该操作
  │    └─ 无权限 → 返回 FAILED, error_code="PERMISSION_DENIED"
  │
  ├─ 4. 参数校验：检查操作参数合法性
  │    ├─ 目标位置在地图范围内？
  │    ├─ 目标实体存在且可见？
  │    ├─ 资源数量为正数？
  │    └─ 不合法 → 返回 FAILED, error_code="INVALID_PARAMS"
  │
  ├─ 5. 游戏规则校验：检查游戏规则约束
  │    ├─ 移动力足够？
  │    ├─ 资源足够？
  │    ├─ 冷却已过？
  │    ├─ 赛季阶段允许该操作？
  │    └─ 不满足 → 返回 FAILED, error_code="RULE_VIOLATION"
  │
  ├─ 6. 幂等校验：检查 request_id 是否已处理
  │    └─ 已处理 → 返回之前的响应（不重复执行）
  │
  ├─ 7. 顺序校验：检查 client_seq 是否连续
  │    └─ 不连续 → 记录警告，仍按收到的顺序处理
  │
  └─ 8. 执行操作 → 返回 SUCCESS + 结果
```

### 错误码表

| 错误码 | 说明 | 客户端处理 |
|--------|------|-----------|
| RATE_LIMIT | 请求频率超限 | 提示"操作过于频繁"，静默丢弃 |
| AUTH_MISMATCH | 身份不匹配 | 断开连接，提示重新登录 |
| PERMISSION_DENIED | 无权限 | 提示"无权限执行此操作" |
| INVALID_PARAMS | 参数不合法 | 提示具体错误（如"目标不在地图范围内"） |
| RULE_VIOLATION | 游戏规则不满足 | 提示具体原因（如"移动力不足"） |
| ENTITY_NOT_FOUND | 目标实体不存在 | 刷新本地状态，提示"目标已消失" |
| RESOURCE_INSUFFICIENT | 资源不足 | 刷新资源显示，提示"资源不足" |
| COOLDOWN_ACTIVE | 冷却中 | 提示剩余冷却时间 |
| SEASON_RESTRICTED | 赛季阶段限制 | 提示"当前阶段不允许此操作" |
| INTERNAL_ERROR | 服务端内部错误 | 提示"服务器错误，请重试" |

## 运营管理操作协议

> 运营管理操作通过独立的AdminService gRPC接口提供，与玩家操作分离。管理操作需要运营管理员Token鉴权+审计日志（见[安全设计.md](../09-服务端架构/安全设计.md)）。

### AdminService gRPC接口

```protobuf
service AdminService {
  // 配置管理
  rpc ReloadConfig(ReloadConfigRequest) returns (ReloadConfigResponse);
  rpc RollbackConfig(RollbackConfigRequest) returns (RollbackConfigResponse);
  
  // 赛季管理
  rpc ForceTransitionPhase(ForceTransitionRequest) returns (ForceTransitionResponse);
  rpc ForceResetSeason(ForceResetRequest) returns (ForceResetResponse);
  
  // 游戏事件管理
  rpc ManualTriggerEvent(TriggerEventRequest) returns (TriggerEventResponse);
  
  // 玩家管理
  rpc BanPlayer(BanPlayerRequest) returns (BanPlayerResponse);
  rpc UnbanPlayer(UnbanPlayerRequest) returns (UnbanPlayerResponse);
  
  // 公告发布
  rpc PublishNotice(PublishNoticeRequest) returns (PublishNoticeResponse);
}
```

### 管理操作消息定义

```protobuf
// 管理操作通用头（所有管理请求携带）
message AdminHeader {
  string admin_token = 1;      // 运营管理员Token
  string admin_id = 2;         // 管理员ID
  string request_id = 3;       // 请求ID（幂等）
  int64 timestamp = 4;         // 时间戳
}

message ReloadConfigRequest {
  AdminHeader header = 1;
  string config_pack_id = 2;   // 配置包ID
  int32 config_version = 3;    // 配置版本
}

message RollbackConfigRequest {
  AdminHeader header = 1;
  int32 target_version = 2;    // 回滚到目标版本
}

message ForceTransitionRequest {
  AdminHeader header = 1;
  string target_phase_id = 2;  // 强制切换到目标阶段
}

message ForceResetRequest {
  AdminHeader header = 1;
  string season_id = 2;
}

message TriggerEventRequest {
  AdminHeader header = 1;
  string event_id = 2;         // 手动触发指定游戏事件
}

message BanPlayerRequest {
  AdminHeader header = 1;
  string player_id = 2;
  int64 duration_seconds = 3;  // 封禁时长（0=永久）
  string reason = 4;
}

message UnbanPlayerRequest {
  AdminHeader header = 1;
  string player_id = 2;
}

message PublishNoticeRequest {
  AdminHeader header = 1;
  string channel = 2;          // 公告频道：world/alliance/private
  string content = 3;
  optional string target_alliance_id = 4;
}
```

### 管理操作鉴权与审计

| 要求 | 说明 |
|------|------|
| 鉴权 | 所有管理操作携带 admin_token，AdminService校验Token有效性+管理员权限（不同管理员有不同操作权限） |
| 审计日志 | 所有管理操作记录审计日志：操作人(admin_id)+操作时间+操作内容+操作结果，append-only不可篡改 |
| 频率限制 | 管理操作频率限制（避免误操作批量触发），如ForceResetSeason 1次/分钟、BanPlayer 10次/分钟 |
| 二次确认 | 危险操作（ForceResetSeason/RollbackConfig）需要二次确认参数 confirm=true |

## 边界条件

| 参数 | 默认值 | 可配 | 说明 |
|------|--------|------|------|
| request_timeout_seconds | 5 | 是 | 客户端请求超时（超时回滚预测） |
| max_request_size_kb | 64 | 是 | 单请求最大大小 |
| max_requests_per_second | 10 | 是 | 每玩家每秒最大请求数（全局限制） |
| request_queue_size | 100 | 是 | 服务端每玩家请求队列大小 |
| response_timeout_seconds | 10 | 是 | 服务端处理超时 |
| push_batch_size | 50 | 是 | 推送消息批量大小 |
| push_interval_ms | 100 | 是 | 推送间隔（非紧急消息） |
| idempotency_cache_size | 1000 | 是 | 幂等缓存大小（最近处理的request_id） |
| client_seq_window | 16 | 是 | 客户端序列号窗口大小（允许的乱序范围） |
